cmake_minimum_required(VERSION 3.23)
project(MoonlightAppleVideo VERSION 0.1.0 LANGUAGES C CXX)
include(CTest)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(MAV_BUILD_TOOLS "Build macOS offline tools" ON)
option(MAV_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(MAV_VT_EXPERIMENTS "Build opt-in VideoToolbox diagnostics (not for production)" OFF)
option(MAV_OPTIMIZED_DEBUG "Optimize the realtime native decoder in Debug builds" ON)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJCXX_STANDARD 23)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
if(MAV_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
set(MAV_PYROWAVE_METAL_DIR "${CMAKE_CURRENT_SOURCE_DIR}/Dependencies/pyrowave/metal")
if(NOT EXISTS "${MAV_PYROWAVE_METAL_DIR}/pyrowave_metal.h")
 message(FATAL_ERROR "Initialize the PyroWave fork: git submodule update --init --recursive")
endif()
add_library(mav-bitstream STATIC src/bitstream.cpp)
target_include_directories(mav-bitstream PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>)
set_target_properties(mav-bitstream PROPERTIES POSITION_INDEPENDENT_CODE ON)
set(MAV_CORE src/decoder.cpp src/clock.cpp src/bitstream.cpp src/pyrowave.cpp Dependencies/pyrowave/metal/pyrowave_bitstream.cpp)
if(APPLE)
 enable_language(OBJCXX)
 list(APPEND MAV_CORE src/backend_vt.mm src/format.mm src/sample.mm src/backend_pyrowave.mm Dependencies/pyrowave/metal/pyrowave_common.mm Dependencies/pyrowave/metal/pyrowave_decoder.mm)
else()
 list(APPEND MAV_CORE src/backend_unavailable.cpp src/pyrowave_unavailable.cpp)
endif()
add_library(moonlight-apple-video STATIC ${MAV_CORE})
add_library(MoonlightAppleVideo::Decoder ALIAS moonlight-apple-video)
set_target_properties(moonlight-apple-video PROPERTIES EXPORT_NAME Decoder POSITION_INDEPENDENT_CODE ON)
target_include_directories(moonlight-apple-video PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}> PRIVATE src Dependencies/pyrowave/metal)
find_package(Threads REQUIRED)
target_link_libraries(moonlight-apple-video PUBLIC Threads::Threads)
# Match SwiftPM's native realtime path: retain debug symbols/assertions while
# avoiding an unoptimized coefficient validator that exceeds high-FPS budgets.
# An empty single-config build type is also unoptimized by default in CMake.
if(MAV_OPTIMIZED_DEBUG AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
 target_compile_options(moonlight-apple-video PRIVATE $<$<OR:$<CONFIG:Debug>,$<CONFIG:>>:-O2>)
endif()
if(APPLE)
 target_link_libraries(moonlight-apple-video PUBLIC "-framework VideoToolbox" "-framework CoreMedia" "-framework CoreVideo" "-framework CoreFoundation" "-framework IOSurface" "-framework Foundation" "-framework Metal" "-framework QuartzCore")
 target_compile_options(moonlight-apple-video PRIVATE $<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>)
 if(MAV_VT_EXPERIMENTS)
  target_compile_definitions(moonlight-apple-video PRIVATE MAV_VT_EXPERIMENTS=1)
 endif()
endif()
if(BUILD_TESTING)

 add_executable(mav-fixture-options-tests tests/fixture_options.cpp)
 target_include_directories(mav-fixture-options-tests PRIVATE tools)
 add_test(NAME fixture-options COMMAND mav-fixture-options-tests)
 add_executable(mav-replay-startup-tests tests/replay_startup.cpp)
 target_include_directories(mav-replay-startup-tests PRIVATE tools)
 add_test(NAME replay-startup COMMAND mav-replay-startup-tests)
 add_executable(mav-bitstream-tests tests/bitstream_tests.cpp)
 target_link_libraries(mav-bitstream-tests PRIVATE mav-bitstream)
 add_test(NAME bitstream COMMAND mav-bitstream-tests)
 add_executable(mav-abi-c tests/abi_c.c)
 target_link_libraries(mav-abi-c PRIVATE moonlight-apple-video)
 add_test(NAME abi-c COMMAND mav-abi-c)
 add_executable(mav-adapter-timing-tests tests/adapter_timing_tests.cpp)
 add_test(NAME adapter-timing COMMAND mav-adapter-timing-tests)
 if(APPLE)
  add_executable(mav-pyrowave-metal-tests tests/pyrowave_metal.mm Dependencies/pyrowave/metal/pyrowave_encoder.mm)
  target_include_directories(mav-pyrowave-metal-tests PRIVATE src Dependencies/pyrowave/metal)

  target_include_directories(mav-pyrowave-benchmark PRIVATE src Dependencies/pyrowave/metal)
  target_compile_options(mav-pyrowave-benchmark PRIVATE -fobjc-arc)
  target_link_libraries(mav-pyrowave-benchmark PRIVATE moonlight-apple-video)
 endif()
 if(APPLE AND MAV_VT_EXPERIMENTS)
  add_executable(mav-vt-controls-tests tests/vt_controls.mm)
  target_include_directories(mav-vt-controls-tests PRIVATE src)
  target_compile_definitions(mav-vt-controls-tests PRIVATE MAV_VT_EXPERIMENTS=1)
  target_compile_options(mav-vt-controls-tests PRIVATE -fobjc-arc)
  target_link_libraries(mav-vt-controls-tests PRIVATE moonlight-apple-video)

 target_compile_options(mav-codec-probe PRIVATE -fobjc-arc)
 foreach(tool fixture replay)
  if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/tools/${tool}.mm)
   add_executable(mav-${tool} tools/${tool}.mm)
   target_include_directories(mav-${tool} PRIVATE src)
